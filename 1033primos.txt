def es_primo(n):
    if n <= 1:
        return False
    
    i = 2
    while i * i <= n:
        if n % i == 0:
            return False
        i += 1
    
    return True

t = int(input())

for _ in range(t):
    a, b = map(int, input().split())
    
    contador = 0
    
    for num in range(a, b + 1):
        if es_primo(num):
            contador += 1
    
    print(contador)
